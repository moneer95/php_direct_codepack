Method 1: JavaScript POST (Recommended)


function proceedToPayment() {
    // Your cartItems array
    const cartItems = [
        { name: "Dental Cleaning", price: 75.00, quantity: 1 },
        { name: "X-Ray Examination", price: 75.00, quantity: 2 },
        { name: "Consultation", price: 50.00, quantity: 1 }
    ];

    // Create form
    const form = document.createElement('form');
    form.method = 'POST';
    form.action = 'index.php';

    // Add cartItems as hidden inputs
    cartItems.forEach((item, index) => {
        ['name', 'price', 'quantity'].forEach(field => {
            const input = document.createElement('input');
            input.type = 'hidden';
            input.name = `cartItems[${index}][${field}]`;
            input.value = item[field];
            form.appendChild(input);
        });
    });

    // Submit form
    document.body.appendChild(form);
    form.submit();
}

Method 3: Fetch API (Alternative)


async function proceedToPayment() {
    const cartItems = [
        { name: "Dental Cleaning", price: 75.00, quantity: 1 },
        { name: "X-Ray Examination", price: 75.00, quantity: 2 },
        { name: "Consultation", price: 50.00, quantity: 1 }
    ];

    const formData = new FormData();
    cartItems.forEach((item, index) => {
        formData.append(`cartItems[${index}][name]`, item.name);
        formData.append(`cartItems[${index}][price]`, item.price);
        formData.append(`cartItems[${index}][quantity]`, item.quantity);
    });

    const response = await fetch('index.php', {
        method: 'POST',
        body: formData
    });

    const html = await response.text();
    document.open();
    document.write(html);
    document.close();
}

function PaymentButton({ cartItems }) {

        const form = document.createElement('form');
        form.method = 'POST';
        form.action = '/index.php';

        cartItems.forEach((item, index) => {
            ['name', 'price', 'quantity'].forEach(field => {
                const input = document.createElement('input');
                input.type = 'hidden';
                input.name = `cartItems[${index}][${field}]`;
                input.value = item[field];
                form.appendChild(input);
            });
        });

        document.body.appendChild(form);
        form.submit();
    };

    return (
        <button onClick={handlePayment}>
            Proceed to Payment
        </button>
    );
}
